describe('Login', () => {
  beforeEach(() => {
    // Acessa a página inicial do BugBank
    cy.visit('https://bugbank.netlify.app/');
  });

  it('Realizar login com credenciais válidas', () => {
    // Pré-condição: Criar um usuário para garantir que as credenciais existam
    // Como o banco é em memória local, precisamos registrar antes de logar
    cy.contains('Registrar').click();
    
    const email = `user${Date.now()}@teste.com`;
    const nome = 'Usuario Teste';
    const senha = '123456';

    cy.get('input[name="email"]').type(email);
    cy.get('input[name="name"]').type(nome);
    cy.get('input[name="password"]').type(senha);
    cy.get('input[name="passwordConfirmation"]').type(senha);
    
    // Clica em Cadastrar
    cy.contains('Cadastrar').click();

    // Aguarda o modal de sucesso (que possui o botão Fechar) e clica para voltar ao login
    // O BugBank geralmente exibe um modal com o texto "foi criada com sucesso"
    cy.contains('Fechar').click();

    // Agora realiza o login com as credenciais criadas
    cy.get('input[name="email"]').type(email);
    cy.get('input[name="password"]').type(senha);
    cy.contains('Acessar').click();

    // Valida se foi redirecionado para a área principal (verificando o texto de saldo ou boas-vindas)
    // O BugBank exibe "Saldo disponível" na tela inicial logada
    cy.contains('Saldo disponível', { timeout: 10000 }).should('be.visible');
  });

  it('Realizar login com senha inválida', () => {
    // Pré-condição: Criar um usuário
    cy.contains('Registrar').click();
    
    const email = `user${Date.now()}@teste.com`;
    const nome = 'Usuario Teste';
    const senha = '123456';

    cy.get('input[name="email"]').type(email);
    cy.get('input[name="name"]').type(nome);
    cy.get('input[name="password"]').type(senha);
    cy.get('input[name="passwordConfirmation"]').type(senha);
    cy.contains('Cadastrar').click();
    cy.contains('Fechar').click();

    // Tenta logar com senha incorreta
    cy.get('input[name="email"]').type(email);
    cy.get('input[name="password"]').type('senha_errada');
    cy.contains('Acessar').click();

    // Valida a mensagem de erro no modal
    cy.contains('Usuário ou senha inválido. Tente novamente ou verifique suas informações!', { timeout: 10000 })
      .should('be.visible');
  });
});

describe('Cadastro de usuário', () => {
  beforeEach(() => {
    // Acessa a página inicial do BugBank
    cy.visit('https://bugbank.netlify.app/');
    // Clica no botão "Registrar" para acessar o formulário de cadastro
    cy.contains('Registrar').click();
  });

  it('Cadastrar novo usuário com dados válidos', () => {
    const email = `user${Date.now()}@teste.com`;
    const nome = 'Usuario Teste';
    const senha = '123456';

    // Preenche o formulário de cadastro
    // Utiliza o atributo 'name' para garantir que o seletor seja único, já que existem múltiplos inputs na página (login e cadastro)
    cy.get('input[name="email"]').type(email);
    cy.get('input[name="name"]').type(nome);
    cy.get('input[name="password"]').type(senha);
    cy.get('input[name="passwordConfirmation"]').type(senha);

    // Solicita a criação da conta
    cy.contains('Cadastrar').click();

    // Valida a mensagem de sucesso no modal
    // O texto contém números dinâmicos (número da conta), então usamos uma expressão regular ou verificação parcial
    cy.contains('A conta', { timeout: 10000 })
      .should('be.visible')
      .and('contain', 'foi criada com sucesso');
    
    // Opcional: Fechar o modal para garantir que o teste finalize limpo
    cy.contains('Fechar').click();
  });

  it('Tentar cadastrar usuário com senhas diferentes', () => {
    const email = `user${Date.now()}@teste.com`;
    const nome = 'Usuario Teste';
    const senha = '123456';
    const senhaDiferente = '654321';

    // Preenche o formulário com senhas divergentes
    cy.get('input[name="email"]').type(email);
    cy.get('input[name="name"]').type(nome);
    cy.get('input[name="password"]').type(senha);
    cy.get('input[name="passwordConfirmation"]').type(senhaDiferente);

    // Solicita a criação da conta
    cy.contains('Cadastrar').click();

    // Valida a mensagem de erro
    // O texto esperado é "As senhas não são iguais"
    cy.contains('As senhas não são iguais', { timeout: 10000 }).should('be.visible');
  });
});

describe('Transferência Bancária', () => {
  // Função auxiliar para criar uma conta e retornar os dados de acesso
  const criarConta = (comSaldo = false) => {
    cy.visit('https://bugbank.netlify.app/');
    cy.contains('Registrar').click();

    const timestamp = Date.now();
    const email = `user${timestamp}@teste.com`;
    const nome = `User ${timestamp}`;
    const senha = '123456';

    cy.get('input[name="email"]').type(email);
    cy.get('input[name="name"]').type(nome);
    cy.get('input[name="password"]').type(senha);
    cy.get('input[name="passwordConfirmation"]').type(senha);

    if (comSaldo) {
      // Ativa o toggle "Criar conta com saldo ?"
      cy.get('#toggleAddBalance').click();
    }

    cy.contains('Cadastrar').click();

    // Captura o número da conta e o dígito a partir da mensagem de sucesso
    // O ID do texto no modal é "modalText"
    return cy.get('#modalText')
      .invoke('text')
      .then((text) => {
        // Regex para extrair número e dígito: "A conta 8-3 foi criada com sucesso"
        const match = text.match(/A conta (\d+)-(\d+)/);
        const accountNumber = match[1];
        const digit = match[2];

        // Fecha o modal usando o ID do botão
        cy.get('#btnCloseModal').click();
        
        // Retorna os dados para uso no teste
        return { email, senha, accountNumber, digit };
      });
  };

  // Realiza login na conta informada
  const realizarLogin = (email, senha) => {
    cy.get('input[name="email"]').type(email);
    cy.get('input[name="password"]').type(senha);
    cy.contains('Acessar').click();
    // Aguarda carregar a home validando o texto de saldo
    cy.contains('Saldo disponível', { timeout: 10000 }).should('be.visible');
  };

  it('Realizar transferência com dados válidos', () => {
    // 1. Criar conta de destino
    criarConta(false).then((contaDestino) => {
      
      // 2. Criar conta de origem COM SALDO
      criarConta(true).then((contaOrigem) => {
        
        // 3. Realizar login na conta de origem
        realizarLogin(contaOrigem.email, contaOrigem.senha);

        // 4. Acessar tela de transferência
        cy.contains('Transferência').click();

        // 5. Preencher dados da transferência
        cy.get('input[name="accountNumber"]').type(contaDestino.accountNumber);
        cy.get('input[name="digit"]').type(contaDestino.digit);
        cy.get('input[name="transferValue"]').type('100');
        cy.get('input[name="description"]').type('Teste de transferência válida');

        // 6. Confirmar transferência
        cy.contains('Transferir agora').click();

        // 7. Validar mensagem de sucesso
        cy.contains('Transferência realizada com sucesso', { timeout: 10000 }).should('be.visible');
        cy.contains('Fechar').click();
      });
    });
  });

  it('Tentar realizar transferência sem saldo suficiente', () => {
    // 1. Criar conta de destino
    criarConta(false).then((contaDestino) => {
      
      // 2. Criar conta de origem SEM SALDO (comSaldo = false)
      criarConta(false).then((contaOrigem) => {
        
        // 3. Realizar login na conta de origem
        realizarLogin(contaOrigem.email, contaOrigem.senha);

        // 4. Acessar tela de transferência
        cy.contains('Transferência').click();

        // 5. Preencher dados com valor superior ao saldo (saldo é 0)
        cy.get('input[name="accountNumber"]').type(contaDestino.accountNumber);
        cy.get('input[name="digit"]').type(contaDestino.digit);
        cy.get('input[name="transferValue"]').type('1000');
        cy.get('input[name="description"]').type('Teste sem saldo');

        // 6. Confirmar transferência
        cy.contains('Transferir agora').click();

        // 7. Validar mensagem de erro
        cy.contains('Você não tem saldo suficiente para essa transação', { timeout: 10000 }).should('be.visible');
        cy.contains('Fechar').click();
      });
    });
  });
});
