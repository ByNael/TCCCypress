describe('Login de usuário - Social Connect', () => {
  const usuario = {
    email: 'test@gmail.com',
    senha: 'password@123',
  };

  const formularioLogin = () => cy.get('form').should('be.visible');

  beforeEach(() => {
    cy.visit('/');
    formularioLogin();
    cy.get('input[name="email"]').should('be.visible');
    cy.get('input[name="password"]').should('be.visible');
  });

  it('Realizar login com credenciais válidas', () => {
    formularioLogin().within(() => {
      cy.get('input[name="email"]')
        .should('be.visible')
        .type(usuario.email);

      cy.get('input[name="password"]')
        .should('be.visible')
        .type(usuario.senha);

      cy.contains('button', /^LOGIN$/)
        .should('be.visible')
        .and('be.enabled')
        .click();
    });

    // Aguarda a autenticação e o carregamento da página inicial.
    cy.get('input[name="email"]').should('not.exist');

    // Confirma que a tela de login foi substituída pelo conteúdo da aplicação.
    cy.get('body').should('be.visible').and(($body) => {
      expect($body.find('input[name="password"]')).to.have.length(0);
    });
  });

  it('Realizar login com senha vazia e apresentar a mensagem required', () => {
    formularioLogin().within(() => {
      cy.get('input[name="email"]')
        .should('be.visible')
        .type(usuario.email);

      cy.get('input[name="password"]')
        .should('be.visible')
        .and('have.value', '');

      cy.contains('button', /^LOGIN$/)
        .should('be.visible')
        .and('be.enabled')
        .click();

      // A validação deve indicar que a senha é obrigatória.
      cy.contains('required')
        .should('be.visible');

      // O usuário deve permanecer na tela de login.
      cy.get('input[name="email"]')
        .should('be.visible')
        .and('have.value', usuario.email);

      cy.get('input[name="password"]')
        .should('be.visible')
        .and('have.value', '');
    });
  });
});

describe('Publicação no feed - Social Connect', () => {
  const usuario = {
    email: 'test@gmail.com',
    senha: 'password@123',
  };

  const publicacao = {
    texto: `Teste E2E de publicação ${Date.now()}`,
    imagem: 'cypress/fixtures/imagem-teste.png',
  };

  beforeEach(() => {
    cy.visit('/');
    cy.get('body').should('be.visible');
  });

  const realizarLogin = () => {
    cy.get('input').should('have.length.greaterThan', 0);

    cy.get('input').then(($inputs) => {
      const campos = [...$inputs];
      const campoEmail = campos.find(
        (input) =>
          input.type === 'email' ||
          /email/i.test(
            `${input.name} ${input.id} ${input.placeholder}`
          )
      );
      const campoSenha = campos.find(
        (input) => input.type === 'password'
      );

      if (campoEmail && campoSenha) {
        cy.wrap(campoEmail).should('be.visible').clear().type(usuario.email);
        cy.wrap(campoSenha).should('be.visible').clear().type(usuario.senha);
      }
    });

    cy.contains('button', /login|entrar|sign in/i)
      .should('be.visible')
      .click();

    cy.get('input[placeholder="Share your memory with a caption..."]', {
      timeout: 15000,
    }).should('be.visible');
  };

  it('Criar uma nova publicação com texto e imagem', () => {
    realizarLogin();

    cy.get('input[placeholder="Share your memory with a caption..."]')
      .should('be.visible')
      .type(publicacao.texto);

    cy.contains('button', 'POST').should('be.disabled');

    cy.get('input[type="file"]', { timeout: 10000 })
      .should('exist')
      .selectFile(publicacao.imagem, { force: true });

    cy.contains('button', 'POST')
      .should('be.visible')
      .and('not.be.disabled');

    cy.get('body').should('contain.text', publicacao.texto);
  });

  it('Publicar conteúdo criado e verificar a publicação no feed', () => {
    realizarLogin();

    cy.get('input[placeholder="Share your memory with a caption..."]')
      .should('be.visible')
      .type(publicacao.texto);

    cy.get('input[type="file"]', { timeout: 10000 })
      .should('exist')
      .selectFile(publicacao.imagem, { force: true });

    cy.contains('button', 'POST')
      .should('be.visible')
      .and('not.be.disabled')
      .click();

    cy.contains(publicacao.texto, { timeout: 15000 })
      .should('be.visible');

    cy.get('img[alt="post"]', { timeout: 15000 })
      .should('exist')
      .and('be.visible');
  });
});
